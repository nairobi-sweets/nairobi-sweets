async function loadProfile(){ let query = sb.from("profiles").select("*").limit(1); if(profileId){ query = query.eq("id", profileId); }else if(slug){ query = query.eq("slug", slug); }else{ app.innerHTML = `
No profile selected.
`; return; } const { data, error } = await query; if(error){ app.innerHTML = `
${error.message}
`; return; } if(!data || data.length === 0){ app.innerHTML = `
Profile not found.
`; return; } const profile = data[0]; const approved = profile.approved === true || profile.approved === "Yes" || profile.approved === "yes" || profile.approved === 1; if(!approved){ app.innerHTML = `
Profile awaiting approval.
`; return; } render(profile); }
