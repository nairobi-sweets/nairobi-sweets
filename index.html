const STORAGE_BUCKET = "profile-images"; function getProfileImage(p) { let raw = String( p.photo_url || p.image_url || p.profile_photo || p.profile_image || p.avatar_url || p.image || p.photo || "" ).trim(); if (!raw || raw.toLowerCase().includes("logo")) { return "assets/logo/logo-badge.png"; } if (raw.startsWith("http://") || raw.startsWith("https://")) { return raw; } raw = raw.replace(/^\/+/, ""); raw = raw.split("/").pop(); return `${SUPABASE_URL}/storage/v1/object/public/${STORAGE_BUCKET}/${raw}`; }
